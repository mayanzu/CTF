import struct
addr = 0x401190
little = struct.pack("<Q", addr)
big = struct.pack(">Q", addr)
print("integer address =", hex(addr))
print("little-endian   =", " ".join("%02x" % b for b in little))
print("big-endian      =", " ".join("%02x" % b for b in big))
print("unpack back     =", hex(struct.unpack("<Q", little)[0]))
