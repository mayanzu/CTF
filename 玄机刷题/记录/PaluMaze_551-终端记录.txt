def generate_maze(size=SIZE, seed=SEED):
    random.seed(seed)
    maze = [[1 for _ in range(size)] for _ in range(size)]
    maze[1][1] = 0
    def carve_path(x, y):

        random.shuffle(directions)

            new_x, new_y = x + dx, y + dy
